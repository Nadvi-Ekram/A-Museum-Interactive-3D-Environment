const marbleVertexShader = `
    varying vec2  vUv;
    varying vec3  vNormal;
    varying vec3  vWorldPosition;

    void main() {
        vUv = uv;
        vNormal = normalize(normalMatrix * normal);
        vec4 wp = modelMatrix * vec4(position, 1.0);
        vWorldPosition = wp.xyz;
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
`;

const marbleFragmentShader = `
    uniform vec3  uLightPos;
    uniform vec3  uLightColor;
    uniform float uTime;

    varying vec2  vUv;
    varying vec3  vNormal;
    varying vec3  vWorldPosition;

    float hash(vec2 p) {
        return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
    }
    float noise(vec2 p) {
        vec2 i = floor(p);
        vec2 f = fract(p);
        float a = hash(i);
        float b = hash(i + vec2(1.0, 0.0));
        float c = hash(i + vec2(0.0, 1.0));
        float d = hash(i + vec2(1.0, 1.0));
        vec2 u = f * f * (3.0 - 2.0 * f);
        return mix(a, b, u.x) + (c - a) * u.y * (1.0 - u.x) + (d - b) * u.x * u.y;
    }

    void main() {
        float n  = noise(vUv * 8.0) * 0.5 + noise(vUv * 20.0) * 0.25;
        float m  = sin(vUv.x * 12.0 + n * 6.0) * 0.5 + 0.5;
        vec3 col = mix(vec3(0.92, 0.90, 0.87), vec3(0.58, 0.55, 0.52), m);

        vec3  L    = normalize(uLightPos - vWorldPosition);
        float diff = max(dot(vNormal, L), 0.0);
        vec3  V    = normalize(cameraPosition - vWorldPosition);
        vec3  H    = normalize(L + V);
        float spec = pow(max(dot(vNormal, H), 0.0), 64.0);

        vec3 ambient  = vec3(0.18);
        vec3 diffuse  = diff * uLightColor;
        vec3 specular = spec * vec3(0.8);

        gl_FragColor = vec4(col * (ambient + diffuse) + specular, 1.0);
    }
`;

const wallVertexShader = `
    varying vec2 vUv;
    varying vec3 vNormal;
    varying vec3 vWorldPosition;
    void main() {
        vUv = uv;
        vNormal = normalize(normalMatrix * normal);
        vec4 wp = modelMatrix * vec4(position, 1.0);
        vWorldPosition = wp.xyz;
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
`;

const wallFragmentShader = `
    uniform vec3 uLightPos;
    uniform vec3 uLightColor;
    varying vec2 vUv;
    varying vec3 vNormal;
    varying vec3 vWorldPosition;

    float hash(vec2 p) {
        return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
    }

    void main() {
        vec2  scaled = vUv * vec2(8.0, 4.0);
        float row    = floor(scaled.y);
        float offset = mod(row, 2.0) * 0.5;
        vec2  cell   = floor(scaled + vec2(offset, 0.0));
        float r      = hash(cell);
        vec3  stone  = mix(vec3(0.52, 0.47, 0.42), vec3(0.72, 0.67, 0.60), r);

        vec2  frac   = fract(scaled + vec2(offset, 0.0));
        float mortar = step(0.04, frac.x) * step(0.06, frac.y);
        stone = mix(vec3(0.28, 0.26, 0.24), stone, mortar);

        vec3  L    = normalize(uLightPos - vWorldPosition);
        float diff = max(dot(vNormal, L), 0.12);

        gl_FragColor = vec4(stone * diff * uLightColor, 1.0);
    }
`;

function makeTexture0() {
    const cv = document.createElement('canvas');
    cv.width = cv.height = 512;
    const c = cv.getContext('2d');

    const sky = c.createLinearGradient(0, 0, 0, 320);
    sky.addColorStop(0, '#04042a'); sky.addColorStop(1, '#0d1b60');
    c.fillStyle = sky; c.fillRect(0, 0, 512, 320);

    for (let i = 0; i < 220; i++) {
        const x = Math.random() * 512, y = Math.random() * 280;
        const r = Math.random() * 2.5 + 0.5;
        const g = c.createRadialGradient(x, y, 0, x, y, r * 4);
        g.addColorStop(0, 'rgba(255,255,210,1)');
        g.addColorStop(1, 'rgba(255,255,210,0)');
        c.fillStyle = g;
        c.beginPath(); c.arc(x, y, r * 4, 0, Math.PI * 2); c.fill();
    }

    const mg = c.createRadialGradient(390, 75, 0, 390, 75, 55);
    mg.addColorStop(0, '#fffacc'); mg.addColorStop(1, '#ffc726');
    c.fillStyle = mg; c.beginPath(); c.arc(390, 75, 55, 0, Math.PI * 2); c.fill();

    for (let i = 0; i < 18; i++) {
        c.strokeStyle = `rgba(80,130,255,${0.15 + Math.random() * 0.2})`;
        c.lineWidth = 2; c.beginPath();
        c.arc(Math.random() * 512, Math.random() * 250, 25 + i * 7, 0, Math.PI * 1.5);
        c.stroke();
    }

    const grd = c.createLinearGradient(0, 300, 0, 512);
    grd.addColorStop(0, '#1b4d1b'); grd.addColorStop(1, '#0e2e0e');
    c.fillStyle = grd; c.fillRect(0, 300, 512, 212);

    c.fillStyle = 'rgba(255,255,200,0.85)';
    c.font = 'italic bold 18px serif';
    c.fillText('Starry Night', 12, 496);
    return new THREE.CanvasTexture(cv);
}

function makeTexture1() {
    const cv = document.createElement('canvas');

function makeTileTexture() {
    const cv = document.createElement('canvas');
    cv.width = cv.height = 512;
    const c = cv.getContext('2d'), ts = 64;
    for (let r = 0; r < 8; r++) {
        for (let col = 0; col < 8; col++) {
            c.fillStyle = (r + col) % 2 === 0 ? '#c4aa80' : '#7a5c34';
            c.fillRect(col * ts, r * ts, ts, ts);
            c.strokeStyle = 'rgba(0,0,0,0.18)'; c.lineWidth = 1;
            c.strokeRect(col * ts + 1, r * ts + 1, ts - 2, ts - 2);
        }
    }
    const t = new THREE.CanvasTexture(cv);
    t.wrapS = t.wrapT = THREE.RepeatWrapping; t.repeat.set(4, 4);
    return t;
}

function makeCeilTexture() {
    const cv = document.createElement('canvas');
    cv.width = cv.height = 256;
    const c = cv.getContext('2d');
    c.fillStyle = '#ede8d8'; c.fillRect(0, 0, 256, 256);
    c.strokeStyle = '#cec4ae'; c.lineWidth = 1.5;
    for (let i = 0; i <= 256; i += 32) {
        c.beginPath(); c.moveTo(i, 0); c.lineTo(i, 256); c.stroke();
        c.beginPath(); c.moveTo(0, i); c.lineTo(256, i); c.stroke();
    }
    const t = new THREE.CanvasTexture(cv);
    t.wrapS = t.wrapT = THREE.RepeatWrapping; t.repeat.set(4, 4);
    return t;
}

function makeWallUniforms(lp) {
    return {
        uLightPos:   { value: lp.clone() },
        uLightColor: { value: new THREE.Vector3(1.0, 0.94, 0.84) }
    };
}

function addWall(w, h, x, y, z, ry) {
    const mat = new THREE.ShaderMaterial({
        vertexShader:   wallVertexShader,
        fragmentShader: wallFragmentShader,
        uniforms: makeWallUniforms(new THREE.Vector3(0, 5, 0)),
        side: THREE.FrontSide
    });
    const mesh = new THREE.Mesh(new THREE.PlaneGeometry(w, h), mat);
    mesh.position.set(x, y, z); mesh.rotation.y = ry;
    mesh.receiveShadow = true; scene.add(mesh);
}

function createDoor() {
    const group   = new THREE.Group();
    const woodMat = new THREE.MeshLambertMaterial({ color: 0x5a2d0c });
    const panelMat = new THREE.MeshLambertMaterial({ color: 0x7a3b12 });
    const knobMat  = new THREE.MeshLambertMaterial({ color: 0xd4af37 });

    // Left frame post
    const lPost = new THREE.Mesh(new THREE.BoxGeometry(0.18, 3.6, 0.18), woodMat);
    lPost.position.set(-0.92, 1.8, 0);
    group.add(lPost);

    // Right frame post
    const rPost = new THREE.Mesh(new THREE.BoxGeometry(0.18, 3.6, 0.18), woodMat);
    rPost.position.set(0.92, 1.8, 0);
    group.add(rPost);

    // Top bar
    const topBar = new THREE.Mesh(new THREE.BoxGeometry(2.02, 0.18, 0.18), woodMat);
    topBar.position.set(0, 3.51, 0);
    group.add(topBar);

    // Door panel
    const panel = new THREE.Mesh(new THREE.BoxGeometry(1.65, 3.3, 0.08), panelMat);
    panel.position.set(0, 1.65, 0);
    group.add(panel);

    // Upper decorative inset
    const insetTop = new THREE.Mesh(new THREE.BoxGeometry(1.3, 1.2, 0.05), woodMat);
    insetTop.position.set(0, 2.6, 0.05);
    group.add(insetTop);

    // Lower decorative inset
    const insetBot = new THREE.Mesh(new THREE.BoxGeometry(1.3, 1.5, 0.05), woodMat);
    insetBot.position.set(0, 0.9, 0.05);
    group.add(insetBot);

    // Door knob
    const knob = new THREE.Mesh(new THREE.SphereGeometry(0.07, 10, 10), knobMat);
    knob.position.set(0.65, 1.65, 0.1);
    group.add(knob);

    // Knob back plate
    const plate = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.05, 0.03, 10), knobMat);
    plate.rotation.x = Math.PI / 2;
    plate.position.set(0.65, 1.65, 0.05);
    group.add(plate);

    // Position on the front wall (z = 10, facing inward)
    group.position.set(0, 0, 9.87);
    group.rotation.y = Math.PI;

    group.traverse(m => { if (m.isMesh) { m.castShadow = true; m.receiveShadow = true; } });
    return group;
}

function createCeilingFixture(x, z) {
    const group = new THREE.Group();

    // Outer housing disc
    const housingMat = new THREE.MeshLambertMaterial({ color: 0xe8e4d8 });
    const housing = new THREE.Mesh(
        new THREE.CylinderGeometry(0.35, 0.35, 0.12, 20),
        housingMat
    );
    group.add(housing);

    // Inner reflector cone
    const reflMat = new THREE.MeshLambertMaterial({ color: 0xd0c8a0, side: THREE.BackSide });
    const reflector = new THREE.Mesh(
        new THREE.ConeGeometry(0.28, 0.18, 16, 1, true),
        reflMat
    );
    reflector.position.y = -0.05;
    group.add(reflector);

    // Glowing bulb sphere
    const bulbMat = new THREE.MeshBasicMaterial({ color: 0xfffde0 });
    const bulb = new THREE.Mesh(new THREE.SphereGeometry(0.14, 14, 14), bulbMat);
    bulb.position.y = -0.08;
    group.add(bulb);

    // Thin rod connecting to ceiling
    const rodMat = new THREE.MeshLambertMaterial({ color: 0x888888 });
    const rod = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.02, 0.3, 8), rodMat);
    rod.position.y = 0.21;
    group.add(rod);

    group.position.set(x, 5.94, z);
    group.traverse(m => { if (m.isMesh) m.castShadow = false; });
    return group;
}

        upright.position.set(ox, 0.925, -0.29);

    const back = new THREE.Mesh(new THREE.BoxGeometry(0.58, 0.62, 0.07), cushMat);
    back.position.set(0, 0.92, -0.27);

    group.rotation.y = ry;
    group.traverse(m => { if (m.isMesh) { m.castShadow = true; m.receiveShadow = true; } });
    scene.add(group);

// =============================================================
//  INPUT

const keys = {};
document.addEventListener('keydown', e => {
    keys[e.key.toLowerCase()] = true;
    if (e.key === '1') swapTexture(0);
    if (e.key === '2') swapTexture(1);
    if (e.key === '3') swapTexture(2);
});
document.addEventListener('keyup', e => { keys[e.key.toLowerCase()] = false; });

document.addEventListener('click', () => {
    activeTex = (activeTex + 1) % paintingTextures.length;
    swapTexture(activeTex);
});

function swapTexture(idx) {
    activeTex = idx;
    paintingMats.forEach((mat, i) => {
        mat.map = paintingTextures[(idx + i) % paintingTextures.length];
        mat.needsUpdate = true;
    });
}

// =============================================================
//  CAMERA MOVEMENT
// =============================================================
let camAngle = 0;
const SPEED  = 0.09;

function updateCamera() {
    if (keys['q']) camAngle -= 0.025;
    if (keys['e']) camAngle += 0.025;
    if (keys['a'] || keys['arrowleft'])  camAngle -= 0.022;
    if (keys['d'] || keys['arrowright']) camAngle += 0.022;

    const fwd = new THREE.Vector3(-Math.sin(camAngle), 0, -Math.cos(camAngle));

    if (keys['w'] || keys['arrowup'])   camera.position.addScaledVector(fwd,  SPEED);
    if (keys['s'] || keys['arrowdown']) camera.position.addScaledVector(fwd, -SPEED);

    camera.position.x = THREE.MathUtils.clamp(camera.position.x, -9, 9);
    camera.position.z = THREE.MathUtils.clamp(camera.position.z, -9, 9);
    camera.position.y = 2;

    camera.lookAt(
        camera.position.x + fwd.x * 10,
        2,
        camera.position.z + fwd.z * 10
    );
}

// =============================================================
//  ANIMATION LOOP
// =============================================================
let t = 0;

function animate() {
    requestAnimationFrame(animate);
    t += 0.016;

    // Spotlight sweeps across statue
    spotlight.position.x = statueX + Math.sin(t * 0.55) * 2.8;
    spotlight.position.z = statueZ + Math.cos(t * 0.35) * 1.5;
    spotlight.target.position.x = statueX + Math.sin(t * 0.55) * 0.4;
    spotlight.target.position.z = statueZ;
    spotlight.target.updateMatrixWorld();

    marbleUniforms.uLightPos.value.set(
        spotlight.position.x,
        spotlight.position.y,
        spotlight.position.z
    );
    marbleUniforms.uTime.value = t;

    updateCamera();
    renderer.render(scene, camera);
}

window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
});

animate();
